\newcommand{\dcdc}[2]{
	\node at ($#1+(-0.3,0.15)$) (#2_in1) {};
	\node at ($#1+(-0.3,-0.15)$) (#2_in2) {};
	\node at ($#1+(0.3,0.15)$) (#2_out1) {};
	\node at ($#1+(0.3,-0.15)$) (#2_out2) {};
	\draw[fill=white,drop shadow] ($#1+(-0.3,-0.3)$) rectangle ($#1+(0.3,0.3)$);
	\draw ($#1+(-0.3,-0.3)$) -- ($#1+(0.3,0.3)$);
	\node at ($#1+(-0.1,0.2)$) () {\tiny\sffamily DC};
	\node at ($#1+(0.1,-0.2)$) () {\tiny\sffamily DC};
}

\newcommand{\fault}[2]{\draw[arrow,red] ($#1!0.4!#2$) to[short] +(0.05,0) -- ($#2+(-0.05,0)$);
	\draw[red] ($#1!0.4!#2$) to[short] +(-0.05,0) to[short] #1;}

\begin{document}
	\begin{tikzpicture}
		\tikzstyle{arrow} = [-{Stealth[length=1.2mm, width=1mm]}]
		
		\node at (0,0) (conv) {};
		\dcdc{(conv)}{C};
		
		\draw (C_in1.base) to[short] +(-0.5,0);
		\draw (C_in2.base) to[short] +(-0.5,0);
		
		\draw (C_out1.base) to[short] +(0.5,0);
		\draw (C_out2.base) to[short] +(0.5,0);
		
		\fault{($(C)+(0.8,0.15)$)}{($(C)+(0.8,-0.15)$)}
		
		\node[anchor=south] at (0,0.7) {\sffamily\textbf{Control}};
		\draw[-{Stealth[length=2mm, width=1.5mm]},line width=1pt] (-0.3,0.6) -- (0.3,0.6);
		
	\end{tikzpicture}
\end{document}
