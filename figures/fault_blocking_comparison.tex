\newcommand{\dcdc}[2]{
	\node at ($#1+(-0.3,0.15)$) (#2_in1) {};
	\node at ($#1+(-0.3,-0.15)$) (#2_in2) {};
	\node at ($#1+(0.3,0.15)$) (#2_out1) {};
	\node at ($#1+(0.3,-0.15)$) (#2_out2) {};
	\draw[fill=white,drop shadow] ($#1+(-0.3,-0.3)$) rectangle ($#1+(0.3,0.3)$);
	\draw ($#1+(-0.3,-0.3)$) -- ($#1+(0.3,0.3)$);
	\node[anchor=northwest] at ($#1+(0.11,0.2)$) () {\tiny\sffamily DC};
	\node[anchor=southeast] at ($#1+(0.41,-0.2)$) () {\tiny\sffamily DC};
}
